```python
import os

# Create the output directory if it doesn't exist
output_dir = 'ttbar_run'
if not os.path.exists(output_dir):
    os.makedirs(output_dir)

# Define the MadGraph5_aMC@NLO command script
command_script = f"""
generate p p > t~ t [QCD]
output {output_dir}
"""

# Write the command script to a file in the output directory
with open(os.path.join(output_dir, 'run_card.dat'), 'w') as f:
    f.write(command_script)
```
