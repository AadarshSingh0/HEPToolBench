```plaintext
import os

# Set output directory
output_directory = 'TTbar_P8_Delphes'
os.makedirs(output_directory, exist_ok=True)

# Change to output directory
os.chdir(output_directory)

# Initialize MadGraph5 with the specified command file
launch -f TTbar_13TeV_pythia8_delphes.cmnd

# Create a new model
import StandardModel

# Define particles
define p = g u c d s u~ c~ d~ s~

# Generate process
generate p p > t t~
add process p p > t~ t

# Set number of events and random seed
set nevents 10000
set iseed 42

# Set beam energies
set ebeam1 6500.
set ebeam2 6500.

# Enable Pythia8 showering
set pythia8_card Shower:QEDshower=on
set pythia8_card Shower:QCDshower=on

# Enable Delphes detector simulation
set delphes_card_path $DELPHES_PATH/cards/delphes_card_ATLAS.tcl

# Disable MadSpin
set madspin_card MadSpin False

# Run the event generation
run

# Finalize and exit
exit
```
